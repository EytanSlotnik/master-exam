\begin{frame}{super quick recap of counting results}
\begin{block}{what we hoped to do}
    Prove or refute there are only exponencily many bounded degree graphs with $O(1)$ average length cycle basis
\end{block}
\begin{block}
    {what we got}
    An exponencial upper bound on bounded degre graphs with $O(1)$ average length weakly fundamental cycle basis
\end{block}

\begin{definition}[Weakly Fundamental Cycle Bases]
    A CB $B$ where we can enumerate the cycles $C_1\ldots C_\nu$ such that there are  $\nu$ edges $e_1\ldots e_\nu$ that hold $e_i\in C_i$ but $e_i\notin C_j$ for $j<i$
\end{definition}
\end{frame}
